cmake_minimum_required(VERSION 3.17.5)
project(rss_cre_probe LANGUAGES C CXX)
find_package(PkgConfig REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CRE_SOURCE "" CACHE PATH "Pinned KOReader CREngine source")
include(engine.cmake)
add_executable(cre-probe probe.cpp)
target_include_directories(cre-probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/cre")
target_link_libraries(cre-probe PRIVATE crengine PkgConfig::CRE_HEADERS)
target_compile_options(cre-probe PRIVATE -Wall -Wextra -UNDEBUG)
pkg_check_modules(BASELINE REQUIRED IMPORTED_TARGET libxml-2.0 freetype2 libpng zlib)
add_executable(baseline baseline.cpp ../../src/feed.cpp ../../src/render.cpp)
target_include_directories(baseline PRIVATE ../../src)
target_link_libraries(baseline PRIVATE PkgConfig::BASELINE)
target_compile_options(baseline PRIVATE -Wall -Wextra -UNDEBUG)
add_executable(page-preview preview.cpp ../../src/device.cpp ../../src/process.cpp)
target_include_directories(page-preview PRIVATE ../../src)
target_link_libraries(page-preview PRIVATE PkgConfig::BASELINE)
target_compile_options(page-preview PRIVATE -Wall -Wextra -UNDEBUG)
if(CMAKE_CROSSCOMPILING)
  target_link_options(cre-probe PRIVATE -static -s)
  target_link_options(baseline PRIVATE -static -s)
  target_link_options(page-preview PRIVATE -static -s)
endif()
