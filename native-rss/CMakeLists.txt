cmake_minimum_required(VERSION 3.17.5)
project(kindle_rss VERSION 0.0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(RSS_DESKTOP "Build SDL desktop simulator" ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XML REQUIRED IMPORTED_TARGET libxml-2.0)
pkg_check_modules(SQLITE REQUIRED IMPORTED_TARGET sqlite3)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(FT REQUIRED IMPORTED_TARGET freetype2)
pkg_check_modules(WEBP REQUIRED IMPORTED_TARGET libwebp)
pkg_check_modules(AVIF REQUIRED IMPORTED_TARGET libavif>=1.1.0)
find_package(Threads REQUIRED)
pkg_check_modules(HOME_CRYPTO REQUIRED IMPORTED_TARGET openssl)
add_executable(kindle-rss-home src/home_bridge.cpp src/home_bridge_main.cpp src/home_patch.cpp)
target_include_directories(kindle-rss-home PRIVATE src)
target_link_libraries(kindle-rss-home PRIVATE PkgConfig::SQLITE PkgConfig::HOME_CRYPTO Threads::Threads)
target_compile_options(kindle-rss-home PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_CROSSCOMPILING)
  target_link_directories(kindle-rss-home PRIVATE ${HOME_CRYPTO_STATIC_LIBRARY_DIRS})
  target_link_libraries(kindle-rss-home PRIVATE ${HOME_CRYPTO_STATIC_LIBRARIES})
  target_link_options(kindle-rss-home PRIVATE -static -s)
endif()
add_library(rss_core src/status.cpp src/feed.cpp src/fulltext.cpp src/store.cpp src/net.cpp src/render.cpp src/app.cpp src/device.cpp src/process.cpp src/image.cpp src/pinyin.cpp src/cre_reader.cpp)
target_include_directories(rss_core PUBLIC src)
target_link_libraries(rss_core PUBLIC PkgConfig::XML PkgConfig::SQLITE PkgConfig::CURL PkgConfig::FT PkgConfig::WEBP PkgConfig::AVIF Threads::Threads)
target_compile_options(rss_core PRIVATE -Wall -Wextra -Wpedantic)
option(RSS_CRE "Use CREngine for article layout" ON)
set(RSS_CRE_SOURCE "" CACHE PATH "Pinned CREngine source checkout")
if(RSS_CRE)
  if(NOT RSS_CRE_SOURCE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/build_reader.py" source-path
                    OUTPUT_VARIABLE RSS_CRE_SOURCE OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE READER_PATH_RESULT)
    if(NOT READER_PATH_RESULT EQUAL 0)
      message(FATAL_ERROR "Unable to read pinned CREngine source path")
    endif()
  endif()
  set(CRE_SOURCE "${RSS_CRE_SOURCE}")
  include(tools/reader.cmake)
  target_compile_definitions(rss_core PRIVATE RSS_CRE=1)
  target_include_directories(rss_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/cre")
  target_link_libraries(rss_core PRIVATE crengine PkgConfig::CRE_HEADERS)
endif()
add_executable(kindle-rss src/main.cpp)
target_link_libraries(kindle-rss PRIVATE rss_core)
if(CMAKE_CROSSCOMPILING AND RSS_CRE)
  target_link_directories(rss_core PUBLIC ${CURL_STATIC_LIBRARY_DIRS} ${AVIF_STATIC_LIBRARY_DIRS})
  target_link_libraries(rss_core PUBLIC ${CURL_STATIC_LIBRARIES} ${AVIF_STATIC_LIBRARIES})
  target_link_options(kindle-rss PRIVATE -static -s)
endif()
if(RSS_DESKTOP)
  pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2)
  target_link_libraries(kindle-rss PRIVATE PkgConfig::SDL)
  target_compile_definitions(kindle-rss PRIVATE RSS_DESKTOP=1)
endif()
include(CTest)
if(BUILD_TESTING)
  find_program(RSS_NODE_EXECUTABLE node)
  if(RSS_NODE_EXECUTABLE)
    add_test(NAME home_card COMMAND ${RSS_NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/home_card_test.js)
  endif()
  add_executable(rss-performance tools/performance.cpp)
  target_link_libraries(rss-performance PRIVATE rss_core)
  foreach(name reader_layout status feed fulltext fulltext_app store render app navigation net device gesture process image pinyin)
    add_executable(${name}_test tests/${name}_test.cpp)
    target_link_libraries(${name}_test PRIVATE rss_core)
    target_compile_options(${name}_test PRIVATE -UNDEBUG)
    add_test(NAME ${name} COMMAND ${name}_test)
  endforeach()
  if(RSS_CRE)
    foreach(name cre_reader cre_app)
      add_executable(${name}_test tests/${name}_test.cpp)
      target_link_libraries(${name}_test PRIVATE rss_core)
      target_compile_options(${name}_test PRIVATE -UNDEBUG)
      add_test(NAME ${name} COMMAND ${name}_test)
    endforeach()
  endif()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME package COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/package_test.py)
    add_test(NAME release COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/release_test.py)
    add_test(NAME launcher COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/launcher_test.py)
    add_test(NAME arm_artifact COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/build_arm_test.py)
    add_test(NAME reader_build COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reader_build_test.py)
    add_test(NAME network COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/network_test.py $<TARGET_FILE:kindle-rss>)
    add_test(NAME home_bridge COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/home_bridge_test.py $<TARGET_FILE:kindle-rss-home>)
    add_test(NAME home_manager COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/home_manager_test.py)
    add_test(NAME home_patch COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/home_patch_test.py $<TARGET_FILE:kindle-rss-home>)
  endif()
endif()
